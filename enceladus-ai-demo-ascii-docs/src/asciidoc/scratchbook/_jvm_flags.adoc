[source,java]
----
import java.math.BigDecimal;
import java.time.Duration;
import java.time.Instant;
import java.util.concurrent.*;

public class Main {
public static void main(String[] args) throws InterruptedException {
int availableProcessors = Runtime.getRuntime().availableProcessors();
System.out.println("Available processors (cores): " + availableProcessors);

        int threadCount = 1_000_000;
        System.out.println("Launching " + threadCount + " virtual threads...");

        ExecutorService executor = Executors.newVirtualThreadPerTaskExecutor();

        CountDownLatch latch = new CountDownLatch(threadCount);
        Instant start = Instant.now();

        for (int i = 0; i < threadCount; i++) {
            executor.execute(() -> {
                // Simulate a CPU-bound task
                BigDecimal swipe = new BigDecimal("110.00");
                BigDecimal apr = new BigDecimal("0.2167"); // 10%
                BigDecimal interest = swipe.multiply(apr);
                BigDecimal outstandingAmount = swipe.add(interest);

                if (ThreadLocalRandom.current().nextInt(1_000_000) < 3) {
                    System.out.printf("Outstanding: $%.2f%n", outstandingAmount);
                }

                latch.countDown();
            });
        }

        latch.await(); // wait for all threads to finish
        executor.close();

        Instant end = Instant.now();
        Duration duration = Duration.between(start, end);

        System.out.println("All tasks completed.");
        System.out.println("Total time taken: " + duration.toMillis() + " ms");
    }
}
----

|===
|JVM Flag| Memory and heap sizing

m|-XX:+UseG1GC                           | Default in Java 21, good for large heaps
m|-XX:MaxHeapFreeRatio=20               | Reduce free heap overhead
m|-XX:MinHeapFreeRatio=5                | Shrink heap aggressively under low usage
m|-XX:+AlwaysPreTouch                   | Prefault memory pages to reduce latency
m|-XX:+UseStringDeduplication           | Useful if many duplicate strings

|GC tuning for throughput|Description

m|-XX:+ParallelRefProcEnabled           | Parallel reference processing
m|-XX:MaxGCPauseMillis=200              | Reasonable pause goal, doesn’t sacrifice throughput
m|-XX:+UnlockDiagnosticVMOptions        | --
m|-XX:+G1UseAdaptiveIHOP                | Let G1 decide Initiating Heap Occupancy percentage
m|-XX:G1ReservePercent=10               | GC reserve buffer
|===

[source,shell]
----
#!/bin/bash

# Usage: ./jvm_monitor.sh <pid> [interval] [count]
# Example: ./jvm_monitor.sh 1234 1 10

PID="$1"
INTERVAL="${2:-1}" # seconds
COUNT="${3:-30}"

if [[ -z "$PID" ]]; then
  echo "Usage: $0 <pid> [interval] [count]"
  exit 1
fi

echo "Monitoring JVM PID $PID every ${INTERVAL}s for $COUNT iterations..."
echo

echo "Legend:"
echo "S0/S1    Survivor spaces"
echo "E        Eden"
echo "O        Old"
echo "M        Metaspace"
echo "YGC/YGCT Young GC count/time"
echo "FGC/FGCT Full GC count/time"
echo "TT      Total GC time"
echo

jstat -gc "$PID" "$INTERVAL" "$COUNT"
----
