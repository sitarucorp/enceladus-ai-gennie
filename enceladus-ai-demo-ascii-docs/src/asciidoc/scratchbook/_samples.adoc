Quick Test: (interest rates are one-to-one link with transaction)
[source,java]
----
    @Order(3)
    @Test
    @DisplayName("""
            3. Calculate accurate total amount due as 207.00 when three transactions passed
            with 100.50 & 99.50 & 200 having variable apr as 3.5, 12 and 27.90
            """)
    void shouldCalculateTotalAmountDueCorrectlyWithVariableApr() {
        Transaction one = Transaction.builder().transactionId("testId1").transactionDate(Instant.now())
                .amount(new BigDecimal("100.50")).aprRate(new BigDecimal("3.5"))
                .userId("FOOOOOO1234").type("SWIPE").build();

        Transaction two = Transaction.builder().transactionId("testId2").transactionDate(Instant.now())
                .amount(new BigDecimal("99.50")).aprRate(new BigDecimal("12.00"))
                .userId("FOOOOOO1234").type("ONLINE").build();

        Transaction three = Transaction.builder().transactionId("testId3").transactionDate(Instant.now())
                .amount(new BigDecimal("200.00")).aprRate(new BigDecimal("27.90"))
                .userId("FOOOOOO1234").type("CASH").build();

        BigDecimal result = TransactionProcessor.calculateTotalDue(List.of(one,two,three));
        BigDecimal minimumDue = TransactionProcessor.calculateMinimumDue(List.of(one,two,three));

        assertEquals("471.26", result.toString());
        assertEquals("23.5630", minimumDue.toString()); // 5% Hard coded (?) to the logic for all plan
    }
